用户、商品、订单、支付表，支持高并发读写。包含分区表与索引优化。
-- 订单汇总查询 (带索引优化)
SELECT c.name, SUM(o.total) AS amount
FROM orders o JOIN customer c ON o.cust_id = c.id
WHERE o.create_time >= '2025-01-01'
GROUP BY c.id HAVING amount > 1000;

学生选课系统
学生、课程、教师、选课关系，包含成绩管理。使用视图简化查询，触发器保证数据一致性。
-- 触发器：选课时检查容量
CREATE TRIGGER check_capacity BEFORE INSERT ON enrollment
FOR EACH ROW
BEGIN
DECLARE cap INT;
SELECT capacity INTO cap FROM course WHERE id=NEW.course_id;
IF (SELECT COUNT(*) FROM enrollment WHERE course_id=NEW.course_id) >= cap THEN
SIGNAL SQLSTATE '45000' SET MESSAGE_TEXT = '课程已满';
END IF;
END;
